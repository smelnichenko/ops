# Kubernetes 1.34.6 -> 1.34.12 (patch): upgrade-kubeadm.yml, one version, drain-free, no kube-proxy addon. kubeadm
# 1.34.12 keeps etcd 3.6.5-0 and CoreDNS v1.12.1 (its own image list). At the rollout setup-kubeadm.yml's
# k8s_package_version becomes 1.34.12-1.1.
# outage: the control-plane pods restart one at a time (the API server away for seconds; workloads keep running), the
# kubelet restarts
# abort: one-way - kubeadm does not downgrade; back only by the etcd backup and a rebuild (task dr:drill's procedure)
image registry.k8s.io/kube-apiserver v1.34.6 => image registry.k8s.io/kube-apiserver v1.34.12
image registry.k8s.io/kube-controller-manager v1.34.6 => image registry.k8s.io/kube-controller-manager v1.34.12
image registry.k8s.io/kube-scheduler v1.34.6 => image registry.k8s.io/kube-scheduler v1.34.12
pkg kubeadm 1.34.6-1.1 (held) => pkg kubeadm 1.34.12-1.1 (held)
pkg kubectl 1.34.6-1.1 (held) => pkg kubectl 1.34.12-1.1 (held)
pkg kubelet 1.34.6-1.1 (held) => pkg kubelet 1.34.12-1.1 (held)
k8s server v1.34.6 => k8s server v1.34.12
playbook upgrade-kubeadm.yml -e k8s_upgrade_to=1.34.12
# the playbook defaults this step moves (scripts/upgrade-defaults.py; production commits them after the step)
default deploy/ansible/playbooks/setup-kubeadm.yml: k8s_package_version: "1.34.6-1.1" => k8s_package_version: "1.34.12-1.1"
