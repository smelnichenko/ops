# versitygw v1.6.0 -> v1.8.0: the in-cluster S3 gateway first (platform upgrade/25-versitygw), then the Pi backup store
# (ops tasks/versitygw.yml; in production outside the 02:00-04:00 backup window). At the rollout the vgw_version
# default becomes 1.8.0. Platform's branch also turns Grafana's Deployment to Recreate (no restart: the pod template
# is unchanged) - Grafana's upgrades (steps 28 and 50) must never run two Grafanas on one SQLite database.
# outage: the in-cluster S3 gateway restarts (Mimir, Tempo and Velero retry their writes); the Pis' versitygw restarts
# one Pi at a time
# abort: revert the step's merge on main (Argo syncs back); the Pis' old package by hand (the no-downgrade check refuses
# the playbook)
image ghcr.io/versity/versitygw v1.6.0 => image ghcr.io/versity/versitygw v1.8.0
pi pi1 versitygw 1.6.0 => pi pi1 versitygw 1.8.0
pi pi2 versitygw 1.6.0 => pi pi2 versitygw 1.8.0
playbook setup-pi-services.yml --tags versitygw -e vgw_version=1.8.0
# A Velero backup after it, and CNPG's WAL archiving and a base backup: both write to the Pi store, whose gateways
# this step restarts (one Pi at a time).
backup-check
barman-check
# and a copy of Postgres recovered from a fresh backup and its WAL (the store Postgres archives to changed)
restore-check
# its change: the branch upgrade/25-versitygw in platform
branch platform
# the playbook defaults this step moves (scripts/upgrade-defaults.py; production commits them after the step)
default deploy/ansible/playbooks/tasks/versitygw.yml: vgw_version: "{{ vgw_version | default('1.6.0') }}" => vgw_version: "{{ vgw_version | default('1.8.0') }}"
